package com.bechuchi.model;

import java.net.DatagramPacket;

/*
 * En klass som ansvarar för basal bearbetning av varje mottaget meddelande,
 * såsom att extrahera och kanske logga viktvärden, IP-adresser osv.
 */
public class MessageProcessor {

    public ClientMessage extractMessageData(DatagramPacket packet) {
        int clientId = 10;
        String ipAddress = packet.getAddress().toString();
        int port = packet.getPort();
        double[] weightValues = { 10.0, 11.2, 13.4, 15.6, 18.9 };

        return new ClientMessage(clientId, ipAddress, port, weightValues);
    }

    public void processMessage(DatagramPacket incomingPacket) {
        try {
            String receivedData = convertByteDataToString(incomingPacket);
            // String latestJsonData;
            // String receivedData = message.convertByteDataToString(incomingPacket);
            System.out.println("Received JSON data: " + receivedData);

            /*
             * synchronized (this) {
             * latestJsonData = receivedData;
             * }
             */
        } catch (Exception e) {
            System.out.println("Error handling client message: " + e.getMessage());
        }
    }

    private String convertByteDataToString(DatagramPacket incomingPacket) {
        String receivedData = new String(incomingPacket.getData(), 0, incomingPacket.getLength());

        return receivedData;
    }

}
